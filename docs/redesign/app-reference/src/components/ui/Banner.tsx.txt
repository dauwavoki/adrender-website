import type { ReactNode } from "react";
import { X, type Icon } from "@phosphor-icons/react";
import { cn } from "../../lib/utils";
import Button from "./Button";
import { renderIconNode } from "./icon";

export default function Banner({
  icon,
  tone = "neutral",
  children,
  action,
  onClose,
  onDismiss,
  className,
}: {
  icon?: ReactNode | Icon;
  tone?: "neutral" | "success" | "danger";
  children: ReactNode;
  action?: ReactNode;
  onClose?: () => void;
  /** Same as onClose. */
  onDismiss?: () => void;
  className?: string;
}) {
  const dismiss = onClose ?? onDismiss;
  const iconNode = renderIconNode(icon, 16, tone === "success" ? "fill" : "regular");
  return (
    <div role="status" className={cn("flex items-center gap-3 rounded-xl bg-inset px-4 py-2.5", className)}>
      {iconNode ? (
        <span
          className={cn(
            "shrink-0",
            tone === "success" && "text-success",
            tone === "danger" && "text-danger",
            tone === "neutral" && "text-secondary",
          )}
        >
          {iconNode}
        </span>
      ) : null}
      <div className="min-w-0 flex-1 text-[13px] leading-[18px] text-secondary">{children}</div>
      {action}
      {dismiss ? (
        <Button variant="ghost" size="sm" iconOnly aria-label="Dismiss" title="Dismiss" onClick={dismiss}>
          <X size={16} aria-hidden />
        </Button>
      ) : null}
    </div>
  );
}
