function quoteErrorMessage(error: unknown): string {
  if (error instanceof Error && error.message) return error.message;
  if (error && typeof error === "object" && "message" in error) {
    const message = (error as { message?: unknown }).message;
    if (typeof message === "string" && message) return message;
  }
  return "request failed";
}

function walkthroughStorageKey(teamId: string): string {
  return teamId ? `${WALKTHROUGH_OPEN_KEY}:${teamId}` : WALKTHROUGH_OPEN_KEY;
}

/** `null` when the user has not chosen. `"1"` open, `"0"` collapsed. */
function readWalkthroughOpen(teamId: string): boolean | null {
  try {
    const value = localStorage.getItem(walkthroughStorageKey(teamId));
    if (value === "1") return true;
    if (value === "0") return false;
    return null;
  } catch {
    return null;
  }
}

function writeWalkthroughOpen(teamId: string, open: boolean) {
  try {
    localStorage.setItem(walkthroughStorageKey(teamId), open ? "1" : "0");
  } catch {
    /* private mode */
  }
}

function relativeTime(iso: string): string {
  const delta = Date.now() - new Date(iso).getTime();
  if (!Number.isFinite(delta) || delta < 0) return "";
  const mins = Math.round(delta / 60000);
  if (mins < 1) return "just now";
  if (mins < 60) return `${mins}m ago`;
  const hours = Math.round(mins / 60);
  if (hours < 24) return `${hours}h ago`;
  const days = Math.round(hours / 24);
  if (days < 30) return `${days}d ago`;
  return new Date(iso).toLocaleDateString(undefined, { month: "short", day: "numeric" });
}

function formatPeriod(iso: string | null | undefined): string | null {
  if (!iso) return null;
  const date = new Date(iso);
  if (Number.isNaN(date.getTime())) return null;
  return date.toLocaleDateString(undefined, { month: "short", day: "numeric", year: "numeric" });
}

function prefersReducedMotion(): boolean {

/** Home under /admin keeps the same screens on the admin prefix. */
function portalPath(pathname: string, path: string): string {
  const base = pathname.startsWith("/admin") ? "/admin" : "/app";
  if (path === "/app" || path.startsWith("/app/") || path.startsWith("/app?")) {
    return base + path.slice(4);

  return path;
}

function HomeBannerHeadline({
  banner,
  quotes,
}: {
  banner: HomeBanner | null;
  quotes: readonly ActiveHomeQuote[];
}) {
  const [shown, setShown] = useState<HomeBanner | null>(null);
  const [outgoing, setOutgoing] = useState<HomeBanner | null>(null);
  const shownAtRef = useRef(0);
  const lastChangeRef = useRef<number | null>(null);
  const fadeTimerRef = useRef<number | null>(null);
  const display = shown ?? banner;

  if (banner && shownAtRef.current === 0) shownAtRef.current = Date.now();

  useEffect(() => {
    if (!banner) return;
    setShown((current) => current ?? banner);
  }, [banner]);

  useEffect(() => {
    return () => {
      if (fadeTimerRef.current != null) window.clearTimeout(fadeTimerRef.current);
    };
  }, []);

  const rotateQuote = () => {
    const now = Date.now();
    const next = pickRotatedHomeQuote({
      current: display,
      quotes,
      shownAt: shownAtRef.current,
      lastChangeAt: lastChangeRef.current,
      now,
    });
    if (!next) return;
    const nextBanner: HomeBanner = { kind: "quote", quote: next.quote, author: next.author };
    if (!prefersReducedMotion()) {
      setOutgoing(display);
      if (fadeTimerRef.current != null) window.clearTimeout(fadeTimerRef.current);
      fadeTimerRef.current = window.setTimeout(() => {
        fadeTimerRef.current = null;
        setOutgoing(null);
      }, QUOTE_FADE_MS);
    }
    setShown(nextBanner);
    shownAtRef.current = now;
    lastChangeRef.current = now;
  };

  return (
    <header className="flex flex-col items-center px-2 py-2 text-center">
      {display ? (
        <div
          className="grid w-full max-w-[960px]"
          onMouseLeave={display.kind === "quote" ? rotateQuote : undefined}
        >
          {outgoing?.kind === "quote" ? (
            <div className="home-quote-out pointer-events-none col-start-1 row-start-1" aria-hidden>
              <QuoteFace banner={outgoing} />
            </div>
          ) : null}
          <div
            key={display.kind === "quote" ? display.quote : display.message}
            aria-live="polite"
            className={cn("col-start-1 row-start-1", outgoing && "home-quote-in")}
          >
            <QuoteFace banner={display} />
          </div>
        </div>
      ) : (
        <div className="flex w-full max-w-2xl flex-col items-center gap-3" aria-hidden>
          <div className="h-10 w-4/5 animate-pulse rounded-full bg-soft" />
          <div className="h-10 w-3/5 animate-pulse rounded-full bg-soft" />
        </div>
      )}
    </header>
  );
}

/**
 * Muted, looping, control-less preview that only plays while on screen. The
 * source attaches the first time the card nears the viewport, so offscreen
 * videos never download. Reduced motion keeps the poster still.
 */
function InViewVideo({ src, poster, className }: { src: string; poster?: string; className?: string }) {
  const wrapRef = useRef<HTMLDivElement>(null);
  const videoRef = useRef<HTMLVideoElement>(null);
  const [attached, setAttached] = useState(false);
  const [playing, setPlaying] = useState(false);
  const [still] = useState(prefersReducedMotion);

  useEffect(() => {
    const el = wrapRef.current;
    if (!el || still || typeof IntersectionObserver === "undefined") return;
    const observer = new IntersectionObserver(
      ([entry]) => {
        const video = videoRef.current;
        if (entry.isIntersecting) {
          setAttached(true);
          if (video && video.src) void video.play().catch(() => {});
        } else if (video) {
          video.pause();
        }
      },
      { threshold: 0.35 },
    );
    observer.observe(el);
    return () => observer.disconnect();
  }, [still]);

  return (
    <div ref={wrapRef} className={cn("relative h-full w-full", className)}>
      <div className="absolute inset-0 transition-transform duration-500 group-hover:scale-105">
        {!still ? (
          <video
            ref={videoRef}
            src={attached ? src : undefined}
            muted
            loop
            playsInline
            autoPlay
            preload="none"
            disablePictureInPicture
            onPlaying={() => setPlaying(true)}
            onPause={() => setPlaying(false)}
            className="pointer-events-none absolute inset-0 h-full w-full object-cover"
          />
        ) : null}
        {poster ? (
          <img
            src={poster}
            alt=""
            loading="lazy"
            className={cn(
              "pointer-events-none absolute inset-0 h-full w-full object-cover transition-opacity duration-300",
              playing ? "opacity-0" : "opacity-100",
            )}
          />
        ) : null}
      </div>
      {!playing ? (
        <div className="pointer-events-none absolute inset-0 flex items-center justify-center">
          <span className="flex size-5 items-center justify-center rounded-full bg-black/55 backdrop-blur-[6px]">
            <Play className="size-3 translate-x-px text-white" weight="fill" />
          </span>
        </div>
      ) : null}
    </div>
  );
}

        if (!cancelled) setPlanLoading(false);

          if (!cancelled) setTokensLoading(false);

      if (document.visibilityState === "visible") load();

    window.addEventListener("focus", onFocus);
    document.addEventListener("visibilitychange", onVisibility);
    return () => {
      cancelled = true;
      window.removeEventListener("focus", onFocus);
      document.removeEventListener("visibilitychange", onVisibility);

  useEffect(() => {
    let cancelled = false;
    bannerRequestRef.current ??= fetchHomeBanner();
    void bannerRequestRef.current
      .then((next) => {
        if (!cancelled) setBanner(next ?? FALLBACK_BANNER);
      })
      .catch((e) => {
        console.warn("[home] banner", e);
        if (!cancelled) setBanner(FALLBACK_BANNER);
      });
    return () => {
      cancelled = true;
    };
  }, []);

  // One read-only fetch per Home visit, after the claimed quote is on screen.

  useEffect(() => {
    if (!banner) return;
    let cancelled = false;
    listRequestRef.current ??= listActiveHomeQuotes(20).catch((error: unknown) => {

        listLoggedRef.current = true;

    void listRequestRef.current.then((rows) => {
      if (!cancelled) setActiveQuotes(rows);
    });
    return () => {
      cancelled = true;

    setStatsExpanded(false);
    setOtherStats(null);
    setOtherLoading(false);
    otherRequestRef.current += 1;

    let cancelled = false;
    setWave2(null);
    void fetchHomeWave2(teamId)

        if (!cancelled) setWave2(data);

      .catch((e) => {
        console.warn("[home] wave 2", e);
        if (!cancelled) {

            stats: { adsGenerated: 0, templatesSaved: 0, brands: 0 },

            continueItem: null,
            hasVideoAd: false,

      cancelled = true;
    };
  }, [teamId]);

    setRecent(null);

        if (!cancelled) setRecent(data);

      cancelled = true;

        .catch((e) => {

          if (!cancelled) setInspired(rows);

      idleId = window.requestIdleCallback(run, { timeout: 800 });
    } else {
      timerId = window.setTimeout(run, 200);

    return () => {
      cancelled = true;
      if (idleId != null) window.cancelIdleCallback(idleId);
      if (timerId != null) window.clearTimeout(timerId);

        setOtherStats({});

          ? "Trial"

                </p>
              ) : isPoolOwner ? (
                <Button
                  variant="secondary"
                  size="sm"
                  onClick={() => setShowAddTokens(true)}
                  leadingIcon={<Plus size={16} />}

                  Add tokens

                Plans and billing

            templateDone={recent?.renderedFromTemplate === true}

            stepsKnown={wave2 != null && recent != null}

                <Rocket size={18} />
              </ColorIconTile>
              <div>
                <h2 className="text-sm font-medium text-primary">Get more done</h2>
                <p className="text-[13px] text-secondary">More renders every month, the moment you need them.</p>

            </div>
            <button
              type="button"
              onClick={() => go("/app/profile?tab=billing")}
              className={cn(linkClass, "inline-flex items-center gap-1")}
            >
              Open billing
              <ArrowRight size={16} />

                    label="Image"
                    action={
                      <button type="button" onClick={() => goTemplates("image")} className={cn(linkClass, "inline-flex items-center gap-1 text-sm")}>
                        View all
                        <ArrowRight size={16} />
                      </button>
                    }
                  >
                    <div className={TEMPLATE_GRID}>
                      {inspired.image.map((card) => (
                        <TemplateCard key={`inspired-${card.id}`} card={card} tone="violet" />
                      ))}
                    </div>
                  </SubRow>
                ) : null}
                {inspired.video.length > 0 ? (
                  <SubRow
                    label="Video"
                    action={
                      <button type="button" onClick={() => goTemplates("video")} className={cn(linkClass, "inline-flex items-center gap-1 text-sm")}>
                        View all
                        <ArrowRight size={16} />
                      </button>
                    }
                  >
                    <div className={TEMPLATE_GRID}>
                      {inspired.video.map((card) => (
                        <TemplateCard key={`inspired-${card.id}`} card={card} tone="amber" />
                      ))}
                    </div>
                  </SubRow>
                ) : null}
              </>
            )}
          </section>
        ) : null}
      </div>
    </div>
  );
}
