/**
 * DESIGN.md §5.1 tokens row. The count string and the Top up handler
 * are passed in — this file does not decide what Top up does.
 * `warning` is the existing low-token condition (balance > 0 and < 20).
 */
import { Coins } from "@phosphor-icons/react";
import Button from "../ui/Button";
import { cn } from "../../lib/utils";

export default function TokensRow({
  collapsed,
  label,
  onTopUp,
  warning = false,
}: {
  collapsed: boolean;
  /** Already formatted, e.g. "91 tokens" or "91 test tokens". */
  label: string;
  onTopUp: () => void;
  warning?: boolean;
}) {
  if (collapsed) {
    return (
      <div className="flex justify-center">
        <Button
          variant="ghost"
          size="md"
          iconOnly
          aria-label={label}
          title={label}
          onClick={onTopUp}
          className={warning ? "text-warning hover:text-warning" : undefined}
        >
          <Coins size={16} aria-hidden />
        </Button>
      </div>
    );
  }

  return (
    <div className="flex h-8 items-center gap-2 px-2.5">
      <Coins size={16} className={cn("shrink-0", warning ? "text-warning" : "text-secondary")} aria-hidden />
      <span
        className={cn(
          "sidebar-label min-w-0 flex-1 whitespace-nowrap text-sm leading-5 tabular-nums",
          warning ? "text-warning" : "text-secondary",
        )}
      >
        {label}
      </span>
      <Button variant="ghost" size="sm" onClick={onTopUp} className="sidebar-label">
        Top up
      </Button>
    </div>
  );
}
