/**
 * DESIGN.md §8.12.
 */
import type { ReactNode } from "react";
import type { Icon } from "@phosphor-icons/react";
import { cn } from "../../lib/utils";
import { renderIconNode } from "./icon";
import { pageTitle } from "./text";

export default function PageHeader({
  icon,
  title,
  subtitle,
  actions,
  className,
  layout = "stack",
}: {
  icon?: ReactNode | Icon;
  title: string;
  subtitle?: ReactNode;
  actions?: ReactNode;
  className?: string;
  /** `row` is the 64px top row (DESIGN.md §8.32). Subtitle stays on that line. */
  layout?: "stack" | "row";
}) {
  const iconNode = renderIconNode(icon, 20);
  const row = layout === "row";
  return (
    <div
      className={cn(
        row
          ? "flex h-16 w-full shrink-0 items-center justify-between gap-4"
          : "mb-6 flex flex-wrap items-start justify-between gap-4",
        className,
      )}
    >
      <div className={cn("min-w-0", row && "flex items-center gap-3")}>
        <div className="flex min-w-0 items-center gap-2">
          {iconNode ? <span className="shrink-0 text-secondary">{iconNode}</span> : null}
          <h1 className={cn(pageTitle, "truncate")}>{title}</h1>
        </div>
        {subtitle ? (
          <p className={cn("text-[13px] leading-[18px] text-secondary", row ? "truncate" : "mt-1")}>
            {subtitle}
          </p>
        ) : null}
      </div>
      {actions ? <div className="flex shrink-0 items-center gap-2">{actions}</div> : null}
    </div>
  );
}
